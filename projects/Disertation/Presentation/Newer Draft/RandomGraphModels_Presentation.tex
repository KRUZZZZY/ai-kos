% ============================================================
%  REFACTORED SLIDE 3 — ER Definition (Simplified)
% ============================================================
\begin{frame}{The Erdős–Rényi Model $G(n,p)$}
  \infobox{%
    \textbf{Definition:} Vertices are connected independently with probability $p$.
  }
  \vspace{10pt}
  
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      {\color{SwanseaBlue}\textbf{Key Metrics:}}
      \begin{itemize}
        \item Expected Edges: $p\binom{n}{2}$
        \item Avg. Degree: $p(n-1)$
        \item Threshold: $p_c = \ln(n)/n$
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \alertbox{%
        \textbf{The Poisson Limit:}\\
        In sparse graphs ($p=\lambda/n$), degrees follow a Poisson distribution.
        \vspace{4pt}
        \textit{Result: No heavy tails or hubs.}
      }
    \end{column}
  \end{columns}
\end{frame}

% ============================================================
%  REFACTORED SLIDE 4 — ER Phase Transitions (Visual focus)
% ============================================================
\begin{frame}[shrink=3]{Erdős–Rényi: Phase Transitions}
  \begin{columns}[T]
    \begin{column}{0.31\textwidth}
      \colorbox{ERBlue}{\parbox{\linewidth}{\centering\color{white}\small\textbf{Subcritical ($\lambda < 1$)}}}
      \begin{itemize}
          \item Small components
          \item Sizes: $O(\log n)$
      \end{itemize}
    \end{column}
    \begin{column}{0.31\textwidth}
      \colorbox{SwanseaGold}{\parbox{\linewidth}{\centering\color{white}\small\textbf{Critical ($\lambda = 1$)}}}
      \begin{itemize}
          \item Giant emerges
          \item Size: $O(n^{2/3})$
      \end{itemize}
    \end{column}
    \begin{column}{0.31\textwidth}
      \colorbox{WSGreen}{\parbox{\linewidth}{\centering\color{white}\small\textbf{Supercritical ($\lambda > 1$)}}}
      \begin{itemize}
          \item Unique giant
          \item Linear size: $O(n)$
      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{15pt}
  \infobox{%
    \textbf{Real-World Gap:} $G(n,p)$ misses the **high clustering** and **heavy tails** seen in actual social or web networks.
  }
\end{frame}

\begin{frame}[shrink=5]{The Watts–Strogatz Small-World Model}
  \infobox{%
    \textbf{Construction:} (1) Begin with a regular ring lattice with $n$ vertices, each
    connected to $k$ nearest neighbours.\ (2) Rewire each edge with probability $\beta$
    to a uniformly chosen random vertex (no self-loops or duplicates).
  }

  \vspace{6pt}
  {\color{SwanseaBlue}\textbf{Effect of rewiring parameter $\beta$:}}
  \vspace{2pt}

  \begin{columns}[T]
    \begin{column}{0.31\textwidth}
      \colorbox{WSGreen}{\parbox{\linewidth}{\centering\color{white}\small\textbf{$\beta=0$}}}\\[2pt]
      {\footnotesize Pure ring lattice: HIGH clustering, LARGE average path length.}
    \end{column}
    \begin{column}{0.31\textwidth}
      \colorbox{SwanseaGold}{\parbox{\linewidth}{\centering\color{white}\small\textbf{$0<\beta\ll 1$}}}\\[2pt]
      {\footnotesize Small-world regime: a few long-range shortcuts reduce $L$ dramatically
      while $C$ stays high.}
    \end{column}
    \begin{column}{0.31\textwidth}
      \colorbox{BAPink}{\parbox{\linewidth}{\centering\color{white}\small\textbf{$\beta=1$}}}\\[2pt]
      {\footnotesize Effectively random: LOW clustering, SHORT path length.}
    \end{column}
  \end{columns}

  \vspace{4pt}
  \begin{center}
    \safeincludegraphics[width=0.9\linewidth, height=2.65cm, keepaspectratio]%
      {ws_beta_sweep_visual}\\[3pt]
    {\tiny\color{MGray}
      $\beta=0$: pure lattice ($C=0.5$, $L=6.63$)
      $\;\to\;$
      $\beta=0.1$: small-world ($C=0.343$, $L=3.64$)
      $\;\to\;$
      $\beta=1$: random ($C=0.096$, $L=2.90$)
    }
  \end{center}
\end{frame}

\begin{frame}[shrink=5]{The Barabási–Albert Preferential Attachment Model}
  \infobox{%
    \textbf{Construction:} (1) Begin with a small initial connected graph on $m_0$
    vertices.\ (2) At each step, add one new vertex with $m\le m_0$ edges.\ (3) Each
    new edge connects to vertex $i$ with probability
    $\Pi(i)=d_i\big/\sum_j d_j$.
  }

  \vspace{2pt}
  {\small\textit{Intuition: ``The rich get richer.'' High-degree vertices attract more
  connections. A positive feedback mechanism.}}

  \vspace{4pt}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{center}
        \safeincludegraphics[width=\linewidth, height=3.45cm, keepaspectratio]%
          {graph_ba_single}
      \end{center}
    \end{column}
    \begin{column}{0.40\textwidth}
      {\color{SwanseaBlue}\textbf{Key structural results:}}\\[4pt]
      \colorbox{BAPink!10}{\parbox{\linewidth}{%
        {\color{BAPink}\small\textbf{Mean degree:}}\\
        {\footnotesize $\langle k\rangle=2m$ \quad (edge-counting argument)}
      }}\\[3pt]
      \colorbox{BAPink!10}{\parbox{\linewidth}{%
        {\color{BAPink}\small\textbf{Power-law tail:}}\\
        {\footnotesize $P(k)\sim 2m^2k^{-3}$ \quad ($\tau=3$, independent of $m$)}
      }}\\[3pt]
      \colorbox{BAPink!10}{\parbox{\linewidth}{%
        {\color{BAPink}\small\textbf{Hubs visible:}}\\
        {\footnotesize $n=80$ BA graph shows nodes with degree 15–22
        alongside most nodes at degree 2–4}
      }}
    \end{column}
  \end{columns}
\end{frame}

% ============================================================
%  REFACTORED SLIDE 8 — BA Scale-Free (Intuition focus)
% ============================================================
\begin{frame}{Barabási–Albert: Scale-Free Networks}
  \begin{center}
    {\Large \color{SwanseaBlue} $P(k) \sim k^{-3}$} \\[5pt]
    \small The "Power Law" Signature
  \end{center}

  \begin{itemize}
    \item \textbf{Heavy Tail:} A few "hubs" hold the vast majority of connections.
    \item \textbf{Scale-Free:} No "typical" node degree (variance is infinite in the limit).
    \item \textbf{Resilience:} Robust to random failure, but fragile under targeted hub attacks.
  \end{itemize}

  \alertbox{%
    \textbf{Why it matters:} Unlike ER or WS, this model explains the "hubs" we see on the World Wide Web and citation networks.
  }
\end{frame}

\begin{frame}{Conclusions: Research Questions Answered}
  \begin{itemize}
    \item \textbf{RQ1 (Phase Transition):} Threshold confirmed at $p \approx 0.0014$. Discrepancy from theory ($0.001$) is due to **finite-size effects**.
    \item \textbf{RQ2 (Small-World):} Short paths form at $\beta=10^{-3}$ while local clustering remains high.
    \item \textbf{RQ3 (Scale-Free):} Power law slope measured at **$-2.73$**; converges to theoretical **$-3$** as $n \to \infty$.
  \end{itemize}

  \vspace{10pt}
  \infobox{%
    \textbf{Takeaway:} No single model is perfect. Real networks require a blend of local clustering (WS) and global hubs (BA).
  }
\end{frame}
